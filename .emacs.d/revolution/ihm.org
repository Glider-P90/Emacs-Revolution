** marginalia
  Rich annotations using the Marginalia package
#+NAME: marginalia
#+BEGIN_SRC emacs-lisp :lexical t

;; ===============================
;; Enable rich annotations using the Marginalia package
;; ===============================
(use-package marginalia
  :ensure t
  ;; Bind `marginalia-cycle' locally in the minibuffer.  To make the binding
  ;; available in the *Completions* buffer, add it to the
  ;; `completion-list-mode-map'.
  :bind (:map minibuffer-local-map
              ("M-A" . marginalia-cycle))
  ;; The :init section is always executed.
  :init
  ;; Marginalia must be activated in the :init section of use-package such that
  ;; the mode gets enabled right away. Note that this forces loading the
  ;; package.
  (marginalia-mode))

#+END_SRC

** Dabbrev
  Abbreviation Completion
#+NAME: Dabbrev
#+BEGIN_SRC emacs-lisp :lexical t

;; ======================================================================
;; Dabbrev: Dynamic Abbreviation
;; ======================================================================
(use-package dabbrev
  :bind (("M-/" . dabbrev-completion)
         ("C-M-/" . dabbrev-expand))
  :config
  ;; Ignore certain buffers/modes
  (add-to-list 'dabbrev-ignored-buffer-regexps "\\` ")
  (dolist (mode '(authinfo-mode doc-view-mode pdf-view-mode tags-table-mode))
    (add-to-list 'dabbrev-ignored-buffer-modes mode))
  ;; Disable ispell completion
  (setq text-mode-ispell-word-completion nil)
  ;; Filter M-x commands (hide irrelevant)
  (setq read-extended-command-predicate #'command-completion-default-include-p))

#+END_SRC

** Corfu
  Completion Framework
#+NAME: corfu
#+BEGIN_SRC emacs-lisp :lexical t

(use-package corfu
  :ensure t
  :custom
  (corfu-cycle t) ;; Cycle through candidates                                                                                                                                                        
  (corfu-quit-at-boundary nil) ;; Don't quit at buffer boundary                                                                                                                                                   
  ;;(corfu-quit-no-match 'separator) ;; Keep popup when no match                                                                                                                                                         
  ;; (corfu-preview-current t) ;; Disable preview                                                                                                                                                                 
  (corfu-preselect 'prompt) ;; Preselect prompt                                                                                                                                                                
  (corfu-auto-prefix 3)	;; Trigger after 3 chars                                                                                                                                                           
  (corfu-auto-trigger "~/.") ;; Trigger on dot                                                                                                                                                                  
  (corfu-separator ?^)
  (corfu-count 10)
  (corfu-min-width 100)
  (global-corfu-minibuffer t)
  ;; (corfu-max-width corfu-min-width)
  :init
  (corfu-echo-mode t)
  (global-corfu-mode)
  (corfu-popupinfo-mode))

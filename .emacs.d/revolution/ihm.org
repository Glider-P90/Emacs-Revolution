** Eglot
  LSP Client
#+NAME: eglot
#+BEGIN_SRC emacs-lisp :lexical t
  ;; ===============================
    ;; Eglot: LSP Client
    ;; ===============================

  (defun eglot--mode-symbol (mode-spec)
    "Retourne le symbole du mode principal pour MODE-SPEC.
  MODE-SPEC peut être :
  - un symbole : python-mode
  - une liste avec des propriétés : (js-mode :language-id \"javascript\")
  - une liste de symboles : (rust-ts-mode rust-mode)"
    (cond
     ((symbolp mode-spec) mode-spec)
     ((and (listp mode-spec) (symbolp (car mode-spec))) (car mode-spec))
     ((listp mode-spec) (car mode-spec))
     (t nil)))

  (use-package eglot
      :ensure t
      :config
      (setq eglot-autoshutdown t
            eglot-send-changes-idle-time 0.3)
      (dolist (entry eglot-server-programs)
        (let ((modes (car entry)))
          (dolist (mode (if (listp modes) modes (list modes)))
            (let ((mode-sym (eglot--mode-symbol mode)))
              (when (and mode-sym (fboundp mode-sym))
                (add-hook (intern (concat (symbol-name mode-sym) "-hook"))
                          'eglot-ensure)))))))

  (use-package lsp-ui
    :after lsp-mode
    :commands lsp-ui-mode)

  ;; ==============================
  ;;  Debuger: Linked LSP
  ;; ==============================
  (use-package dap-mode
    :ensure t
    :init
    (dap-mode 1))

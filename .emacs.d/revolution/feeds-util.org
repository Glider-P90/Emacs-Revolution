** Globals

List of RSS feeds for Emacs Revolution.
Each entry :
+ :name     (string)  — feed display name
+ :url      (string)  — feed URL
+ :verified (boolean) — whether the feed has been verified.
+ :level    (integer) — priority level to read

#+BEGIN_EXMPLE

(setq my_list '((:name "Lilo" :url "https://lilo.org/feed" :verified t)))

#+END_EXMPLE

#+NAME: Feeds
#+BEGIN_SRC emacs-lisp :lexical t

  (defcustom er-feeds-list nil
    "List of RSS feeds for Emacs Revolution."
    :type '(repeat
  	  (plist
  	   :name (string :tag "Name")
  	   :url (string :tag "URL")
  	   :verified (boolean :tag "Verified")
  	   :level (integer :tag "Level")))
    :group 'emacs-revolution)
#+END_SRC

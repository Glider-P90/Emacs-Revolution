  (defun validate-url (url-er &rest ignore)
    "Validate url. Return URL only if http(s)://, else nil."
    (if (and (stringp url-er) (> (length url-er) 0)
  	   (string-match-p "https?://" url-er))
        url-er
      nil))

  (defun rss-form-filter (rss-form &rest ignore)
    "Add Feed in Emacs-Revolution from WIDGET."
    (let* ((entries (widget-get rss-form :children))
  	 (new-entry (car (reverse entries)))
  	 (last-entry (car (cdr (reverse entries))))
  	 (feed last-entry)
  	 (total (length entries)))
      (when (length> entries 1)
        (if-let* ((feed-url (validate-url (string-trim (car (cdr (widget-value feed))))))
  		(feed-name (if (and (stringp feed-url) (stringp (car (widget-value feed)))
  				    (= (length (car (widget-value feed))) 0))
  			       (get-domain feed-url)
  			     (car (widget-value feed)))))
  	  (progn (message "Adding feed %s: %s" feed-name feed-url)
  		 (widget-editable-list-delete-at rss-form last-entry)
  		 (previous-line 1)
  		 (beginning-of-line)
  		 (widget-create 'group
  				(widget-create 'push-button
  					       :tag "Retirer"
  					       :action (lambda (widget &rest ignore)
  							 (widget-default-delete (widget-get widget :parent)))
  					       )
  				(widget-create 'url-link
  					       :button-prefix "[ "
  					       :button-suffix " ]"
  					       :tag feed-name
  					       feed-url
  					       )
  				)
  		 )
  	(progn (message "Warning: %s url is not valid, please correct it." feed-url)
  	       (widget-editable-list-delete-at rss-form new-entry)
  	       )
  	))
      )
    )

  (defun form-installer ()
    "Display form's install."
    (interactive)
    (let ((dir-dep-widget)
  	(auto-maj-widget)
  	(rss-list-widget)
  	(widget-example-repeat))
      (save-excursion
        (goto-char (- (buffer-size) (length "\n\n[Installer]")))
        (setq auto-maj-widget
  	    (widget-create 'radio-button-choice
  			   :format "\nPlannification des mise a jour automatique?\n%[%v%]\n\t************************\n"
  			   :validate (lambda (widget &rest ignore)
  				       (if (string= (widget-value widget) "Oui")
  					   (er-update-is-enable t)
  					 (er-update-is-enable nil)))
  			   :value "Oui"
  			   '(item "Oui") '(item "Non")))
        
        (setq rss-list-widget
  	    (widget-create 'editable-list
  			   :tag "rss"
  			   :value '("titi")
  			   :entry-format "\n%d %v"
  			   :format "\nSouhaitez vous ajouter des flux RSS a suivre?\n%i%v\n\n\t************************\n"
  			   :notify #'rss-form-filter
  			   rss-editable-line
  			   ))
        (widget-setup)
        (setq install-current-step 1))))

  (defun welcome-widget ()
    "Welcome to install."
    (interactive)
    (setq inhibit-startup-screen t)
    (let ((inhibit-read-only t))
      (switch-to-buffer "*Bienvenue sur Emacs Revolution*")
      (set-buffer "*Bienvenue sur Emacs Revolution*")
      (widget-insert (concat "Bienvenue sur Emacs Revolution "
        			   emacs-revolution-version
        			   "\n\nChoisir le mode d'installation :\n"))
      (setq install-mode-widget
        	  (widget-create 'radio-button-choice
        			 :value "default"
        			 :notify (lambda (widget &rest ignore)
        				   (when (string= (widget-value install-mode-widget) "custom")
        				     (form-installer)
        				     (message "Choice install [%s] mode" (widget-value widget))))
        			 '(item "default") '(item "custom")))
      (widget-insert "\n\t\t")
      (widget-create 'push-button
        		   :value "Installer"
        		   :button-face '(:foreground "brightwhite" :background "black")
        		   :value-face '(:foreground "brightwhite" :background "black")
        		   :notify (lambda (widget &rest ignore)
        			     (if (string= (widget-value widget) "Installer")
        				 (install-emacs-revolution))))
      (use-local-map widget-keymap)
      (widget-setup)
      (when (get-buffer "*GNU Emacs*")
        (kill-buffer "*GNU Emacs*"))
      (when (get-buffer "*scratch*")
        (kill-buffer "*scratch*"))
      ))

  (add-hook 'after-init-hook #'welcome-widget)

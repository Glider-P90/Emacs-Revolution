#+NAME: Shortcuts
#+BEGIN_SRC emacs-lisp :lexical t

(use-package which-key
  :ensure t
  :custom
  (which-key-show-early-on-C-h t)
  (which-key-idle-delay 10000)
  (which-key-idle-secondary-delay 0.05)
  (which-key-side-window-max-width 0.40)
  (which-key-separator " → ")
  (which-key-prefix-prefix "+")
  (which-key-sort-order 'which-key-local-then-key-order)
  (which-key-max-description-length 60)
  (which-key-show-docstrings t))

(global-unset-key (kbd "<f3>"))
(global-unset-key (kbd "<f4>"))
(global-unset-key (kbd "<f5>"))

;; Annuler/Refaire
(global-set-key [(control z)] 'undo)

;;******     Insert an header      ******;;
(global-set-key (kbd "C-c h") 'std-file-header)

;;******  Affiche le mode majeur   ******;;
(global-set-key (kbd "C-c m") 'show-major-mode)

;; Documentation du langage
(global-set-key (kbd "<f2>") #'er-describe-symbol-at-point)

;; debuger deplacer depuis f4 en f5 pour une logique d'utilisation
(let ((cmd (lookup-key global-map (kbd "<f4>"))))
  (when (commandp cmd)
    (global-unset-key (kbd "<f4>"))
    (global-set-key (kbd "<f5>") cmd)))

;; Documentation des raccourcis
(global-set-key (kbd "<f3>") #'projectile-help-command)
(global-set-key (kbd "<f4>") #'which-key-show-top-level)

;; Navigation des pages UNIQUEMENT quand which-key est visible
(define-key which-key-mode-map (kbd "C-<right>") #'which-key-show-next-page)
(define-key which-key-mode-map (kbd "C-<left>")  #'which-key-show-previous-page)

#+END_SRC

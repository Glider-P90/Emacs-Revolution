;; ===============================
;; ag: Outil de recherche de texte ultra-rapide.
;; ===============================
(use-package ag
  :ensure t)

;; ===============================
;; which-key: Affiche les raccourcis clavier possibles dans le minibuffer.
;; ===============================
(use-package which-key
  :ensure t
  :config
  (which-key-mode +1))

;; ===============================
;; neotree: Explorateur de fichiers sous forme d'arbre.
;; ===============================
(use-package neotree
  :bind ([f8] . neotree-toggle)
  :config (setq neo-default-system-application "open"))
    
;; ===============================
;; Projectile: Gestionnaire de projets, naviguer, rechercher et exécuter des commandes.
;; ===============================
(use-package projectile
  :ensure t
  :init
  (setq projectile-project-search-path '("~/Projects/" ))
  :config
  (define-key projectile-mode-map (kbd "C-c C-p") 'projectile-command-map)
  (global-set-key (kbd "C-c p") 'projectile-command-map)
  (projectile-mode +1))

;; ===============================
;; flycheck: Outil de vérification syntaxique.
;; ===============================
(use-package flycheck-
  :ensure t
  :init (global-flycheck-mode))

(use-package flycheck-posframe
  :ensure t
  :after flycheck
  :hook (flycheck-mode . flycheck-posframe-mode))

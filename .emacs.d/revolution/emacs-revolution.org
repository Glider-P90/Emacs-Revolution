(defgroup emacs-revolution nil
  "Emacs Revolution."
  :group 'tools)

(defcustom er-version "0.7b"
  "This variable indicate the project version of Emacs Revolution."
  :type 'string
  :group 'emacs-revolution)

(add-hook 'emacs-startup-hook
          (lambda ()
            (message "Emacs Revolution version %s loaded on %2.3fs mode." er-version
                     (float-time
    		      (time-subtract after-init-time before-init-time)))))

(defcustom emacs-revolution-dir
  (expand-file-name "~/.emacs.d/revolution/")
  "Root of the project Emacs-Revolution."
  :type 'directory
  :group 'emacs-revolution)

(defcustom dependencies-project-revolution
  (lambda ()
    (setq e-r-dependencies (concat (getenv "HOME") "/.emacs.d/dependencies"))
    (if (file-exists-p e-r-dependencies)
	(message "Emacs-revolution Dependencies already installed.")
      (make-directory (concat (getenv "HOME") "/.emacs.d/dependencies"))))
  "Set up folders and config files for Emacs Revolution."
  :type 'function
  :group 'emacs-revolution)
(fset 'dependencies-project-revolution dependencies-project-revolution)

(defcustom lang-settings-dir (expand-file-name "revolution/modes/" user-emacs-directory)
  "Folder containing the configuration files by major-mode."
  :type 'directory
  :group 'emacs-revolution)

(defcustom er-update-is-enable nil
  "The update can be set on with t ether nil disable updates."

  :group 'emacs-revolution)

(defcustom er-auto-package-update-interval 7

(defcustom er-auto-package-update-prompt-before-update t

  :group 'emacs-revolution)

(defcustom er-auto-package-update-hide-results nil

(defcustom er-mode-found nil
  "The variable er-mode-found permet de signaler un changement de mode."
  :type 'boolean
  :group 'emacs-revolution)

(defcustom er-mode-src (symbol-name major-mode)
  "The variable er-mode-src permet de reprendre le mode d'origine."
  :type 'string
  :group 'emacs-revolution)

(defcustom er-major-mode-origin major-mode
  "Main Major mode from opened file used as origin when major-mode change."
  :type 'symbol
  :group 'emacs-revolution)

(defcustom set-er-major-mode-origin (lambda ()
  			     (setq er-major-mode-origin major-mode)
  			     (message "[E-R] Set Major Mode origin: %s." er-major-mode-origin))
  "Get main major mode of the opened file."
  :type 'function
  :group 'emacs-revolution)
(fset 'set-er-major-mode-origin set-er-major-mode-origin)

(defcustom emacs-revolution-major-mode-hook ()
  "Hook: Emacs revolution setting up major mode."
  :type 'hook
  :group 'emacs-revolution)

(add-hook 'emacs-revolution-major-mode-hook #'set-er-major-mode-origin)

(unless er-major-mode-origin
  (setq er-major-mode-origin major-mode))

** Packages source and update
#+NAME: Packages source and update
#+BEGIN_SRC emacs-lisp :lexical t

(package-initialize)
(unless (package-installed-p 'use-package)
  (when (er-update-is-enable)
    (package-refresh-contents))
  (package-install 'use-package))
(eval-and-compile
    (setq use-package-always-ensure t
          use-package-expand-minimally t))

(use-package auto-package-update
  :custom
  (auto-package-update-interval er-auto-package-update-interval)
  (auto-package-update-prompt-before-update er-auto-package-update-prompt-before-update)
  (auto-package-update-hide-results er-auto-package-update-hide-results)
  :config
  (auto-package-update-maybe)
  (auto-package-update-at-time er-auto-package-update-at-time))

(use-package quelpa
  :ensure t)

(use-package quelpa-use-package
  :ensure t
  :after quelpa
  :config
  (quelpa-use-package-activate-advice))

(defun revolution-load-org-config()
  "[E-R] Info: Load revolution configuration."
  (org-babel-load-file "~/.emacs.d/revolution/ihm.org")

      (progn (message "Emacs Start Revolution.")

  (org-babel-load-file "~/.emacs.d/revolution/header.org")

  (org-babel-load-file "~/.emacs.d/revolution/shortcuts.org")

    (defun get-rss-entries (widget)
    "Extraire les entrées du editable-list en une liste de plist."
    (let ((entries (widget-value widget))
          result)
      (dolist (entry entries result)
        (when (and entry (listp entry))
          (push (list (car entry)  (cadr entry)) result)))))

    (defun form-install-display ()
      "Display form's install."
      (interactive)
      (let ((dir-dep-widget)
    	(auto-maj-widget)
    	(rss-list-widget)
    	(widget-example-repeat))
        (save-excursion
          (goto-char (- (buffer-size) (length "\n\n[Installer]")))
          (setq dir-dep-widget
    	    (widget-create 'directory
    			   :format (concat "\n\n ***   Completion avec M-TAB   ***\n\nDefault Path: "
    					   emacs-revolution-path-dependancies
    					   "\nDirectory: %v\n\n\t************************\n")
    			   :error "Please, enter a directory path for dependancies."
    			   :value emacs-revolution-path-dependancies
    			   :size 25
    			   :notify (lambda (widget &rest ignore)
    				     (setq emacs-revolution-path-dependancies (widget-value widget)))))
          (setq auto-maj-widget
    	    (widget-create 'radio-button-choice
    			   :format "\nPlannification des mise a jour automatique?\n%[%v%]\n\t************************\n"
    			   :validate (lambda (widget &rest ignore)
    				       (if (string= (widget-gevalue widget) "Oui")
    					   (er-update-is-enable t)
    					 (er-update-is-enable nil)))
      			   :value "Oui"
  			   '(item "Oui") '(item "Non")))
  	(setq rss-list-widget
  	      (widget-create 'editable-list
  			     :format "\nSouhaitez vous ajouter des flux RSS a suivre?\n%i%v\n\n\t************************\n"
  			     :entry-format "\n%d %v"
  			     :value '("")

  			       (let ((count (length (widget-get widget :children)))
  				     (rss-widget-children (reverse (widget-get widget :children))))
  				 (when (> count 1)
  				   (while rss-widget-children

  					       (progn
  						 (widget-editable-list-delete-at widget (car (last rss-widget-children)))
  						 (message "Please, complete fields before add an other RSS."))

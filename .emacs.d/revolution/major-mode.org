** Show Major Mode
#+NAME: show_major_mode
#+BEGIN_SRC emacs-lisp :lexical t

;;; Permet d'afficher le mode majeur actuel
(defun show-major-mode ()
  "Affiche le mode majeur pendant 5 secondes."
  (interactive)
  (let ((content (format "Mode majeur : %s" major-mode))
        (buffer-name "*Major Mode*"))
    (with-current-buffer (get-buffer-create buffer-name)
      (erase-buffer)
      (insert content))
    (let ((window (display-buffer buffer-name
        			  '((display-buffer-reuse-window
        			     display-buffer-below-selected)
        			    (window-height . 2)))))
      (run-at-time "5 sec" nil #'delete-window window)
      )))

#+END_SRC

** Major Mode Loading
#+NAME: Major-Mode-Loading
#+BEGIN_SRC emacs-lisp :lexical t

(defun register-auto-modes ()
  "Ajoute les associations à `auto-mode-alist` et charge le mode si nécessaire."
  (dolist (entry mode-associations)
    (let* ((pattern (car entry))
           (mode    (cdr entry))
           (file-mode (concat lang-settings-dir (symbol-name mode) ".el")))
      (when (and buffer-file-name
                 (string-match-p pattern buffer-file-name))
        (when (file-exists-p file-mode)
          (load-file file-mode)
          (funcall mode)))
      (add-to-list 'auto-mode-alist (cons pattern mode)))))

(defun load-lang-settings ()
  "Charge dynamiquement le fichier de configuration associé au major mode courant."
  (if (not (eq er-major-mode-origin major-mode))
      (when buffer-file-name
        (let* ((er-major-mode-origin major-mode)
               (er-src-mode-name (symbol-name er-major-mode-origin))
               (file-mode (concat lang-settings-dir (concat er-src-mode-name ".el"))))
          (if (file-exists-p file-mode)
              (progn (message "[E-R] Info: major mode [%s] witch file %s" (symbol-name er-major-mode-origin) file-mode)
          	(load-file file-mode))
            )))))

(defun lang-settings-init ()
  "Initialise les réglages de langage pour Emacs Revolution."
  (register-auto-modes)
  (load-lang-settings)
  (run-hooks 'emacs-revolution-major-mode-hook))


(lang-settings-init)

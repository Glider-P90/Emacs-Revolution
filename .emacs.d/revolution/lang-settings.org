(defun register-auto-modes ()
  "Ajoute les associations à `auto-mode-alist` et charge le mode si nécessaire."
  (dolist (entry mode-associations)
    (let* ((pattern (car entry))
           (mode    (cdr entry))
           (file-mode (concat lang-settings-dir (symbol-name mode) ".el")))
      (when (and buffer-file-name
                 (string-match-p pattern buffer-file-name))
        (when (file-exists-p file-mode)
          (load-file file-mode)
          (funcall mode)))
      (add-to-list 'auto-mode-alist (cons pattern mode)))))

(defun load-lang-settings ()
  "Charge dynamiquement le fichier de configuration associé au major mode courant."
  (if (not (eq er-src-mode major-mode))
      (when buffer-file-name
      	(let* ((er-src-mode major-mode)
               (er-src-mode-name (symbol-name er-src-mode))
               (file-mode (concat lang-settings-dir (concat er-src-mode-name ".el"))))
      	  (if (file-exists-p file-mode)
              (progn
      		(load-file file-mode))
      	    )))))

(defun lang-settings-init ()
  "Initialise les réglages de langage pour Emacs Revolution."
  (register-auto-modes)
  (add-hook 'after-change-major-mode-hook 'load-lang-settings))

(lang-settings-init)
